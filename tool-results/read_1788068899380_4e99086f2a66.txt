    34→function latLngToVec3(lat: number, lng: number, radius: number): THREE.Vector3 {
    35→  const phi = (90 - lat) * (Math.PI / 180);
    36→  const theta = (lng + 180) * (Math.PI / 180);
    37→  return new THREE.Vector3(
    38→    -radius * Math.sin(phi) * Math.cos(theta),
    39→    radius * Math.cos(phi),
    40→    radius * Math.sin(phi) * Math.sin(theta),
    41→  );
    42→}

    76→
    77→  /* React state for UI overlays */

    79→  const [containerW, setContainerW] = useState(400);
    80→  const [zoomedCountry, setZoomedCountry] = useState<GlobeCountryData | null>(null);
    81→  const [textureLoaded, setTextureLoaded] = useState(false);
    82→

    87→
    88→  /* ────────────────────────────────────────────────────────────────────
    89→     Three.js Initialization — runs once, immediately on mount
    90→     ──────────────────────────────────────────────────────────────────── */

    92→    const container = canvasWrapRef.current;
    93→    if (!container) return;
    94→
    95→    /* Check WebGL */
    96→    const testCanvas = document.createElement('canvas');
    97→    const gl = testCanvas.getContext('webgl2') || testCanvas.getContext('webgl');
    98→    if (!gl) return;
    99→
   100→    const w = container.clientWidth;
   101→    const h = container.clientHeight;
   102→    if (w < 1 || h < 1) return;
   103→
   104→    /* ── Scene ── */
   105→    const scene = new THREE.Scene();
   106→
   107→    /* ── Camera ── */
   108→    const camera = new THREE.PerspectiveCamera(45, w / h, 0.1, 100);
   109→    camera.position.set(0, 0.2, 2.6);
   110→    camera.lookAt(0, 0, 0);
   111→
   112→    /* ── Renderer ── */
   113→    const renderer = new THREE.WebGLRenderer({
   114→      antialias: true,
   115→      alpha: true,
   116→      powerPreference: 'high-performance',
   117→    });
   118→    renderer.setSize(w, h);
   119→    renderer.setPixelRatio(Math.min(window.devicePixelRatio, 2));
   120→    renderer.setClearColor(0x000000, 0); // transparent
   121→    container.appendChild(renderer.domElement);
   122→
   123→    /* ── Lights ── */
   124→    const ambientLight = new THREE.AmbientLight(0xffffff, 1.2);
   125→    scene.add(ambientLight);
   126→
   127→    const sunLight = new THREE.DirectionalLight(0xfff8ee, 1.8);
   128→    sunLight.position.set(5, 3, 5);
   129→    scene.add(sunLight);
   130→
   131→    const fillLight = new THREE.DirectionalLight(0x8ecae6, 0.4);
   132→    fillLight.position.set(-4, -1, -3);
   133→    scene.add(fillLight);
   134→
   135→    /* ── Earth Group (rotated together) ── */
   136→    const earthGroup = new THREE.Group();
   137→    earthGroup.rotation.x = 0.25;
   138→    earthGroup.rotation.y = -0.5;
   139→    scene.add(earthGroup);
   140→
   141→    /* ── Earth Sphere ── */
   142→    const earthGeom = new THREE.SphereGeometry(1, 64, 48);
   143→
   144→    /* Start with a bright, visible blue placeholder */
   145→    const placeholderMat = new THREE.MeshPhongMaterial({
   146→      color: 0x2277bb,
   147→      emissive: 0x112244,
   148→      shininess: 25,
   149→    });
   150→    const earthMesh = new THREE.Mesh(earthGeom, placeholderMat);
   151→    earthGroup.add(earthMesh);
   152→
   153→    /* ── Atmosphere glow (outer, BackSide) ── */
   154→    const atmosGeom = new THREE.SphereGeometry(1.04, 48, 36);
   155→    const atmosMat = new THREE.ShaderMaterial({
   156→      vertexShader: `
   157→        varying vec3 vNormal;
   158→        varying vec3 vPosition;
   159→        void main() {
   160→          vNormal = normalize(normalMatrix * normal);
   161→          vPosition = (modelViewMatrix * vec4(position, 1.0)).xyz;
   162→          gl_Position = projectionMatrix * modelViewMatrix * vec4(position, 1.0);
   163→        }
   164→      `,
   165→      fragmentShader: `
   166→        varying vec3 vNormal;
   167→        varying vec3 vPosition;
   168→        void main() {
   169→          vec3 viewDir = normalize(-vPosition);
   170→          float rim = 1.0 - max(0.0, dot(viewDir, vNormal));
   171→          float glow = pow(rim, 3.0) * 1.2;
   172→          gl_FragColor = vec4(0.05, 0.58, 0.53, glow * 0.35);
   173→        }
   174→      `,
   175→      transparent: true,
   176→      side: THREE.BackSide,
   177→      depthWrite: false,
   178→    });
   179→    scene.add(new THREE.Mesh(atmosGeom, atmosMat));
   180→
   181→    /* ── Markers Group ── */
   182→    const markersGroup = new THREE.Group();
   183→    earthGroup.add(markersGroup);
   184→
   185→    /* ── Store refs ── */
   186→    const threeCtx = {
   187→      renderer,
   188→      scene,
   189→      camera,
   190→      earthGroup,
   191→      earthMesh,
   192→      markersGroup,
   193→      rafId: 0,
   194→      disposed: false,
   195→    };

   197→
   198→    /* ── Load Earth Texture ── */
   199→    const textureLoader = new THREE.TextureLoader();
   200→
   201→    function applyTexture(tex: THREE.Texture) {
   202→      if (threeCtx.disposed) return;
   203→      tex.colorSpace = THREE.SRGBColorSpace;
   204→      tex.anisotropy = renderer.capabilities.getMaxAnisotropy();
   205→      const realMat = new THREE.MeshStandardMaterial({
   206→        map: tex,
   207→        roughness: 0.75,
   208→        metalness: 0.05,
   209→      });
   210→      earthMesh.material = realMat;
   211→      placeholderMat.dispose();
   212→      setTextureLoaded(true);
   213→    }
   214→
   215→    /* Try local HD texture first */
   216→    textureLoader.load(
   217→      '/textures/earth-day-hd.jpg',
   218→      (tex) => {
   219→        applyTexture(tex);

   221→      undefined,
   222→      () => {
   223→        /* Fallback to SD local */
   224→        textureLoader.load(
   225→          '/textures/earth-day.jpg',
   226→          (tex) => {
   227→            applyTexture(tex);
   228→          },
   229→          undefined,
   230→          () => {
   231→            /* Fallback to CDN */
   232→            textureLoader.load(
   233→              'https://unpkg.com/three-globe/example/img/earth-day.jpg',
   234→              (tex) => {
   235→                applyTexture(tex);
   236→              },
   237→              undefined,
   238→              () => {
   239→                /* All texture sources failed, keeping placeholder */
   240→                setTextureLoaded(true);
   241→              },
   242→            );
   243→          },
   244→        );
   245→      },
   246→    );
   247→
   248→    /* ── Animation Loop ── */
   249→    function animate() {
   250→      if (threeCtx.disposed) return;
   251→      threeCtx.rafId = requestAnimationFrame(animate);
   252→
   253→      /* Auto-rotate */
   254→      if (autoRotate.current && !isDragging.current && !zoomingToCountry.current) {
   255→        earthGroup.rotation.y += 0.001;
   256→      }
   257→
   258→      /* Momentum */
   259→      if (!isDragging.current && !autoRotate.current && !zoomingToCountry.current) {
   260→        const m = momentum.current;
   261→        if (Math.abs(m.x) > 0.0001 || Math.abs(m.y) > 0.0001) {
   262→          earthGroup.rotation.y += m.x;
   263→          earthGroup.rotation.x = Math.max(-0.8, Math.min(0.8, earthGroup.rotation.x + m.y));
   264→          m.x *= 0.92;
   265→          m.y *= 0.92;
   266→        }
   267→      }
   268→
   269→      /* Smooth zoom */
   270→      camera.position.z += (zoomTarget.current - camera.position.z) * 0.06;
   271→      camera.position.y += (0.2 - camera.position.y) * 0.04;
   272→
   273→      /* Zoom-to-country rotation */
   274→      if (zoomingToCountry.current && targetRotY.current !== null) {
   275→        let diff = targetRotY.current - earthGroup.rotation.y;
   276→        while (diff > Math.PI) diff -= Math.PI * 2;
   277→        while (diff < -Math.PI) diff += Math.PI * 2;
   278→        earthGroup.rotation.y += diff * 0.04;
   279→        if (targetRotX.current !== null) {
   280→          earthGroup.rotation.x += (targetRotX.current - earthGroup.rotation.x) * 0.04;
   281→        }
   282→      }
   283→
   284→      /* Marker visibility (hide when on back side) */
   285→      for (let i = 0; i < markersGroup.children.length; i++) {
   286→        const child = markersGroup.children[i];
   287→        if (child.userData.isGlow || child.userData.isRing) continue;
   288→        const wPos = new THREE.Vector3();
   289→        child.getWorldPosition(wPos);
   290→        const normal = wPos.clone().normalize();
   291→        const toCamera = camera.position.clone().sub(wPos).normalize();
   292→        child.visible = toCamera.dot(normal) > -0.15;
   293→      }
   294→
   295→      renderer.render(scene, camera);
   296→    }
   297→    threeCtx.rafId = requestAnimationFrame(animate);
   298→
   299→    /* ── Resize handler ── */
   300→    function onResize() {
   301→      if (threeCtx.disposed) return;
   302→      const cw = container.clientWidth;
   303→      const ch = container.clientHeight;
   304→      if (cw < 1 || ch < 1) return;
   305→      camera.aspect = cw / ch;
   306→      camera.updateProjectionMatrix();
   307→      renderer.setSize(cw, ch);
   308→      setContainerW(cw);
   309→    }
   310→    window.addEventListener('resize', onResize);
   311→    setContainerW(w);
   312→
   313→    /* ── Cleanup ── */
   314→    return () => {
   315→      threeCtx.disposed = true;
   316→      cancelAnimationFrame(threeCtx.rafId);
   317→      window.removeEventListener('resize', onResize);
   318→      if (idleTimer.current) clearTimeout(idleTimer.current);
   319→
   320→      /* Dispose Three.js resources */
   321→      earthGeom.dispose();
   322→      atmosGeom.dispose();
   323→      atmosMat.dispose();
   324→      renderer.dispose();
   325→
   326→      if (container.contains(renderer.domElement)) {
   327→        container.removeChild(renderer.domElement);
   328→      }
   329→      threeRef.current = null;
   330→    };
   331→  }, []);
   332→
   333→  /* ────────────────────────────────────────────────────────────────────
   334→     Build markers when countries data arrives
   335→     ──────────────────────────────────────────────────────────────────── */
   336→  useEffect(() => {
   337→    const ctx = threeRef.current;
   338→    if (!ctx) return;
   339→    const { markersGroup } = ctx;
   340→
   341→    /* Clear old markers */
   342→    while (markersGroup.children.length > 0) {
   343→      const child = markersGroup.children[0];
   344→      markersGroup.remove(child);
   345→      child.traverse((obj) => {
   346→        if ((obj as THREE.Mesh).geometry) (obj as THREE.Mesh).geometry.dispose();
   347→        if ((obj as THREE.Mesh).material) {
   348→          const mat = obj as THREE.Mesh;
   349→          if (mat.material instanceof THREE.SpriteMaterial) mat.material.map?.dispose();
   350→          (mat.material as THREE.Material).dispose();
   351→        }
   352→      });
   353→    }
   354→    markerMap.current.clear();
   355→
   356→    /* Create new markers */
   357→    for (const country of activeCountries) {
   358→      const pos = latLngToVec3(country.latitude, country.longitude, 1.005);
   359→      const isHL = country.isHighlighted;
   360→
   361→      /* Pin group */
   362→      const pin = new THREE.Group();
   363→      const stemH = isHL ? 0.06 : 0.04;
   364→      const headR = isHL ? 0.018 : 0.012;
   365→
   366→      const stem = new THREE.Mesh(
   367→        new THREE.CylinderGeometry(0.003, 0.005, stemH, 8),
   368→        new THREE.MeshBasicMaterial({ color: isHL ? 0x07c2b8 : 0x0d9488, transparent: true, opacity: 0.9 }),
   369→      );
   370→      stem.position.y = stemH / 2;
   371→      pin.add(stem);
   372→
   373→      const head = new THREE.Mesh(
   374→        new THREE.SphereGeometry(headR, 12, 12),
   375→        new THREE.MeshBasicMaterial({ color: isHL ? 0x07c2b8 : 0x0d9488 }),
   376→      );
   377→      head.position.y = stemH + headR * 0.3;
   378→      pin.add(head);
   379→
   380→      pin.position.copy(pos);
   381→      pin.userData.isMarker = true;
   382→      markersGroup.add(pin);
   383→      markerMap.current.set(head.id, country);
   384→
   385→      /* Glow sprite */
   386→      const glowCanvas = document.createElement('canvas');
   387→      glowCanvas.width = 64;
   388→      glowCanvas.height = 64;
   389→      const gCtx = glowCanvas.getContext('2d')!;
   390→      const grad = gCtx.createRadialGradient(32, 32, 0, 32, 32, 32);
   391→      grad.addColorStop(0, isHL ? 'rgba(7,194,184,0.9)' : 'rgba(13,148,136,0.7)');
   392→      grad.addColorStop(0.4, isHL ? 'rgba(7,194,184,0.2)' : 'rgba(13,148,136,0.15)');
   393→      grad.addColorStop(1, 'rgba(0,0,0,0)');
   394→      gCtx.fillStyle = grad;
   395→      gCtx.fillRect(0, 0, 64, 64);
   396→
   397→      const glowSprite = new THREE.Sprite(
   398→        new THREE.SpriteMaterial({
   399→          map: new THREE.CanvasTexture(glowCanvas),
   400→          transparent: true,
   401→          depthWrite: false,
   402→          blending: THREE.AdditiveBlending,
   403→        }),
   404→      );
   405→      glowSprite.position.copy(pos);
   406→      glowSprite.position.y += stemH + headR * 0.5;
   407→      glowSprite.scale.set(isHL ? 0.12 : 0.08, isHL ? 0.12 : 0.08, 1);
   408→      glowSprite.userData.isGlow = true;
   409→      markersGroup.add(glowSprite);
   410→    }
   411→  }, [activeCountries]);
   412→
   413→  /* ────────────────────────────────────────────────────────────────────
   414→     Pointer interaction handlers
   415→     ──────────────────────────────────────────────────────────────────── */
   416→  useEffect(() => {
   417→    const el = canvasWrapRef.current;
   418→    if (!el) return;
   419→
   420→    let dragDist = 0;
   421→
   422→    const onDown = (e: PointerEvent) => {
   423→      isDragging.current = true;
   424→      autoRotate.current = false;
   425→      dragDist = 0;
   426→      prevPointer.current = { x: e.clientX, y: e.clientY };
   427→      momentum.current = { x: 0, y: 0 };
   428→      el.setPointerCapture(e.pointerId);
   429→      if (idleTimer.current) clearTimeout(idleTimer.current);
   430→      /* Cancel zoom-to if dragging */
   431→      if (zoomingToCountry.current) {
   432→        zoomingToCountry.current = null;
   433→        targetRotY.current = null;
   434→        targetRotX.current = null;
   435→        zoomTarget.current = 2.6;
   436→        setZoomedCountry(null);
   437→      }
   438→    };
   439→
   440→    const onMove = (e: PointerEvent) => {
   441→      if (isDragging.current) {
   442→        const dx = e.clientX - prevPointer.current.x;
   443→        const dy = e.clientY - prevPointer.current.y;
   444→        dragDist += Math.abs(dx) + Math.abs(dy);
   445→        const g = threeRef.current?.earthGroup;
   446→        if (g) {
   447→          g.rotation.y += dx * 0.004;
   448→          g.rotation.x = Math.max(-0.8, Math.min(0.8, g.rotation.x + dy * 0.004));
   449→          momentum.current = { x: dx * 0.002, y: dy * 0.001 };
   450→        }
   451→        prevPointer.current = { x: e.clientX, y: e.clientY };
   452→        setTooltip(null);
   453→        el.style.cursor = 'grabbing';
   454→        return;
   455→      }
   456→
   457→      /* Hover detection via raycasting */
   458→      const ctx = threeRef.current;
   459→      if (!ctx) return;
   460→      const rect = el.getBoundingClientRect();
   461→      const nx = ((e.clientX - rect.left) / rect.width) * 2 - 1;
   462→      const ny = -((e.clientY - rect.top) / rect.height) * 2 + 1;
   463→      const raycaster = new THREE.Raycaster();
   464→      raycaster.setFromCamera(new THREE.Vector2(nx, ny), ctx.camera);
   465→
   466→      const meshes: THREE.Mesh[] = [];
   467→      for (const marker of ctx.markersGroup.children) {
   468→        if (!marker.userData.isMarker || !marker.visible) continue;
   469→        marker.traverse((child) => {
   470→          if ((child as THREE.Mesh).isMesh) meshes.push(child as THREE.Mesh);
   471→        });
   472→      }
   473→      const hits = raycaster.intersectObjects(meshes, false);
   474→
   475→      if (hits.length > 0) {
   476→        let obj: THREE.Object3D | null = hits[0].object;
   477→        while (obj && !obj.userData.isMarker) obj = obj.parent;
   478→        if (obj) {
   479→          const headMesh = obj.children[1] as THREE.Mesh;
   480→          const country = markerMap.current.get(headMesh.id);
   481→          if (country) {
   482→            setTooltip({ country, x: e.clientX - rect.left, y: e.clientY - rect.top });
   483→            el.style.cursor = 'pointer';
   484→            return;
   485→          }
   486→        }
   487→      }
   488→      setTooltip(null);
   489→      el.style.cursor = 'grab';
   490→    };
   491→
   492→    const onUp = () => {
   493→      isDragging.current = false;
   494→      el.style.cursor = 'grab';
   495→      idleTimer.current = setTimeout(() => { autoRotate.current = true; }, 4000);
   496→    };
   497→
   498→    const onWheel = (e: WheelEvent) => {
   499→      e.preventDefault();
   500→      zoomTarget.current = Math.max(1.5, Math.min(5, zoomTarget.current + e.deltaY * 0.0006));
   501→    };
   502→
   503→    const onClick = (e: MouseEvent) => {
   504→      if (dragDist > 8) return;
   505→      const ctx = threeRef.current;
   506→      if (!ctx) return;
   507→      const rect = el.getBoundingClientRect();
   508→      const nx = ((e.clientX - rect.left) / rect.width) * 2 - 1;
   509→      const ny = -((e.clientY - rect.top) / rect.height) * 2 + 1;
   510→      const raycaster = new THREE.Raycaster();
   511→      raycaster.setFromCamera(new THREE.Vector2(nx, ny), ctx.camera);
   512→
   513→      const meshes: THREE.Mesh[] = [];
   514→      for (const marker of ctx.markersGroup.children) {
   515→        if (!marker.userData.isMarker || !marker.visible) continue;
   516→        marker.traverse((child) => {
   517→          if ((child as THREE.Mesh).isMesh) meshes.push(child as THREE.Mesh);
   518→        });
   519→      }
   520→      const hits = raycaster.intersectObjects(meshes, false);
   521→
   522→      if (hits.length > 0) {
   523→        let obj: THREE.Object3D | null = hits[0].object;
   524→        while (obj && !obj.userData.isMarker) obj = obj.parent;
   525→        if (obj) {
   526→          const headMesh = obj.children[1] as THREE.Mesh;
   527→          const country = markerMap.current.get(headMesh.id);
   528→          if (!country) return;
   529→          targetRotY.current = -country.longitude * (Math.PI / 180);
   530→          targetRotX.current = country.latitude * (Math.PI / 180) * -0.4;
   531→          zoomTarget.current = country.isHighlighted ? 1.7 : 1.9;
   532→          zoomingToCountry.current = country.id;
   533→          setZoomedCountry(country);
   534→          setTooltip(null);
   535→        }
   536→      }
   537→    };
   538→
   539→    /* Touch pinch zoom */
   540→    let lastPinchDist = 0;
   541→    const onTouchStart = (e: TouchEvent) => {
   542→      if (e.touches.length === 2) {
   543→        const dx = e.touches[0].clientX - e.touches[1].clientX;
   544→        const dy = e.touches[0].clientY - e.touches[1].clientY;

   546→      }

   551→        const dy = e.touches[0].clientY - e.touches[1].clientY;
   552→        const dist = Math.sqrt(dx * dx + dy * dy);
   553→        const delta = lastPinchDist - dist;
   554→        zoomTarget.current = Math.max(1.5, Math.min(5, zoomTarget.current + delta * 0.005));
   555→        lastPinchDist = dist;
   556→      }

   560→    el.addEventListener('pointermove', onMove);
   561→    el.addEventListener('pointerup', onUp);

   563→    el.addEventListener('wheel', onWheel, { passive: false });

   570→      el.removeEventListener('pointermove', onMove);
   571→      el.removeEventListener('pointerup', onUp);
   572→      el.removeEventListener('pointerleave', onUp);
   573→      el.removeEventListener('wheel', onWheel);
   574→      el.removeEventListener('click', onClick);

   578→    };
   579→  }, []);
