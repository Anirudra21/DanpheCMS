    84→    84→
    85→    85→function latLngToVec3(lat: number, lng: number, radius: number): THREE.Vector3 {
    86→    86→  const phi = (90 - lat) * (Math.PI / 180);
    87→    87→  const theta = (lng + 180) * (Math.PI / 180);
    88→    88→  return new THREE.Vector3(
    89→    89→    -radius * Math.sin(phi) * Math.cos(theta),
    90→    90→    radius * Math.cos(phi),
    91→    91→    radius * Math.sin(phi) * Math.sin(theta),
    92→    92→  );
    93→    93→}

    98→    98→
    99→    99→function createEarthTexture(): THREE.CanvasTexture {
   100→   100→  const W = 2048;
   101→   101→  const H = 1024;
   102→   102→  const canvas = document.createElement('canvas');
   103→   103→  canvas.width = W;
   104→   104→  canvas.height = H;
   105→   105→  const ctx = canvas.getContext('2d')!;
   106→   106→
   107→   107→  // Ocean
   108→   108→  ctx.fillStyle = '#091a28';
   109→   109→  ctx.fillRect(0, 0, W, H);
   110→   110→
   111→   111→  // Subtle ocean gradient (lighter at equator)
   112→   112→  const oceanGrad = ctx.createLinearGradient(0, 0, 0, H);
   113→   113→  oceanGrad.addColorStop(0, 'rgba(15,35,55,0.5)');
   114→   114→  oceanGrad.addColorStop(0.5, 'rgba(20,45,65,0.3)');
   115→   115→  oceanGrad.addColorStop(1, 'rgba(10,25,40,0.5)');
   116→   116→  ctx.fillStyle = oceanGrad;
   117→   117→  ctx.fillRect(0, 0, W, H);
   118→   118→
   119→   119→  // Grid
   120→   120→  ctx.strokeStyle = 'rgba(255,255,255,0.025)';
   121→   121→  ctx.lineWidth = 1;
   122→   122→  for (let lng = -180; lng <= 180; lng += 30) {
   123→   123→    const x = ((lng + 180) / 360) * W;
   124→   124→    ctx.beginPath(); ctx.moveTo(x, 0); ctx.lineTo(x, H); ctx.stroke();
   125→   125→  }
   126→   126→  for (let lat = -90; lat <= 90; lat += 30) {
   127→   127→    const y = ((90 - lat) / 180) * H;
   128→   128→    ctx.beginPath(); ctx.moveTo(0, y); ctx.lineTo(W, y); ctx.stroke();
   129→   129→  }
   130→   130→
   131→   131→  // Draw landmasses
   132→   132→  for (const polygon of LAND) {
   133→   133→    ctx.beginPath();
   134→   134→    for (let i = 0; i < polygon.length; i++) {
   135→   135→      const [lat, lng] = polygon[i];
   136→   136→      const x = ((lng + 180) / 360) * W;
   137→   137→      const y = ((90 - lat) / 180) * H;
   138→   138→      if (i === 0) ctx.moveTo(x, y);
   139→   139→      else ctx.lineTo(x, y);
   140→   140→    }
   141→   141→    ctx.closePath();
   142→   142→    ctx.fillStyle = '#0f2a38';
   143→   143→    ctx.fill();
   144→   144→    ctx.strokeStyle = 'rgba(13,148,136,0.18)';
   145→   145→    ctx.lineWidth = 1.5;
   146→   146→    ctx.stroke();
   147→   147→  }
   148→   148→
   149→   149→  // Subtle coastal glow pass
   150→   150→  for (const polygon of LAND) {
   151→   151→    ctx.beginPath();
   152→   152→    for (let i = 0; i < polygon.length; i++) {
   153→   153→      const [lat, lng] = polygon[i];
   154→   154→      const x = ((lng + 180) / 360) * W;
   155→   155→      const y = ((90 - lat) / 180) * H;
   156→   156→      if (i === 0) ctx.moveTo(x, y);
   157→   157→      else ctx.lineTo(x, y);
   158→   158→    }
   159→   159→    ctx.closePath();
   160→   160→    ctx.strokeStyle = 'rgba(13,148,136,0.08)';
   161→   161→    ctx.lineWidth = 4;
   162→   162→    ctx.stroke();
   163→   163→  }
   164→   164→
   165→   165→  // Noise dots for texture feel
   166→   166→  ctx.fillStyle = 'rgba(255,255,255,0.008)';
   167→   167→  for (let i = 0; i < 6000; i++) {
   168→   168→    const x = Math.random() * W;
   169→   169→    const y = Math.random() * H;
   170→   170→    ctx.fillRect(x, y, 1, 1);
   171→   171→  }
   172→   172→
   173→   173→  const tex = new THREE.CanvasTexture(canvas);
   174→   174→  tex.colorSpace = THREE.SRGBColorSpace;
   175→   175→  return tex;
   176→   176→}

   181→   181→
   182→   182→function createGlowTexture(r: number, g: number, b: number): THREE.CanvasTexture {
   183→   183→  const s = 128;
   184→   184→  const c = document.createElement('canvas');
   185→   185→  c.width = s; c.height = s;
   186→   186→  const ctx = c.getContext('2d')!;
   187→   187→  const grad = ctx.createRadialGradient(s / 2, s / 2, 0, s / 2, s / 2, s / 2);
   188→   188→  grad.addColorStop(0, `rgba(${r},${g},${b},0.9)`);
   189→   189→  grad.addColorStop(0.3, `rgba(${r},${g},${b},0.3)`);
   190→   190→  grad.addColorStop(0.7, `rgba(${r},${g},${b},0.05)`);
   191→   191→  grad.addColorStop(1, 'transparent');
   192→   192→  ctx.fillStyle = grad;
   193→   193→  ctx.fillRect(0, 0, s, s);
   194→   194→  return new THREE.CanvasTexture(c);
   195→   195→}

   200→   200→
   201→   201→const ATMO_VS = `
   202→   202→  varying vec3 vWorldNormal;
   203→   203→  varying vec3 vViewPos;
   204→   204→  void main() {
   205→   205→    vec4 wp = modelMatrix * vec4(position, 1.0);
   206→   206→    vViewPos = (viewMatrix * wp).xyz;
   207→   207→    vWorldNormal = normalize(mat3(modelMatrix) * normal);
   208→   208→    gl_Position = projectionMatrix * viewMatrix * wp;
   209→   209→  }
   210→   210→`;
   211→   211→
   212→   212→const ATMO_FS = `
   213→   213→  varying vec3 vWorldNormal;
   214→   214→  varying vec3 vViewPos;
   215→   215→  void main() {
   216→   216→    vec3 viewDir = normalize(-vViewPos);
   217→   217→    float rim = 1.0 - max(0.0, dot(viewDir, vWorldNormal));
   218→   218→    float glow = pow(rim, 3.5) * 1.4;
   219→   219→    gl_FragColor = vec4(0.051, 0.58, 0.533, glow * 0.45);
   220→   220→  }
   221→   221→`;

   245→   245→  const idleTimerRef = useRef<ReturnType<typeof setTimeout> | null>(null);

   248→   248→  const zoomingToRef = useRef<string | null>(null);
   249→   249→  const targetRotYRef = useRef<number | null>(null);
   250→   250→
   251→   251→  // Raycasting

   253→   253→  const markerMapRef = useRef<Map<number, GlobeCountryData>>(new Map());
   254→   254→
   255→   255→  // React state
   256→   256→  const [tooltip, setTooltip] = useState<{ country: GlobeCountryData; x: number; y: number } | null>(null);

   262→   262→  const totalHospitals = useMemo(() => activeCountries.reduce((s, c) => s + c.hospitalCount, 0), [activeCountries]);
   263→   263→  const nepalData = useMemo(() => countries.find((c) => c.isHighlighted) || null, [countries]);
   264→   264→
   265→   265→  // ─── Three.js scene setup (runs once when in view) ──────────────────
   266→   266→  useEffect(() => {
   267→   267→    const container = containerRef.current;

   270→   270→    const h = container.clientHeight;

   290→   290→    rendererRef.current = renderer;
   291→   291→

   294→   294→    const dir = new THREE.DirectionalLight(0xffffff, 0.9);
   295→   295→    dir.position.set(5, 3, 5);
   296→   296→    scene.add(dir);

   298→   298→    rim.position.set(-4, -2, -4);
   299→   299→    scene.add(rim);
   300→   300→
   301→   301→    // Earth group (rotation target)
   302→   302→    const earthGroup = new THREE.Group();
   303→   303→    earthGroup.rotation.x = 0.25; // slight tilt
   304→   304→    earthGroup.rotation.y = -0.5;
   305→   305→    scene.add(earthGroup);
   306→   306→    earthGroupRef.current = earthGroup;
   307→   307→

   309→   309→    const earthTex = createEarthTexture();
   310→   310→    const earthGeom = new THREE.SphereGeometry(EARTH_RADIUS, 80, 80);
   311→   311→    const earthMat = new THREE.MeshPhongMaterial({
   312→   312→          map: earthTex,
   313→   313→          shininess: 12,
   314→   314→          specular: new THREE.Color('#0e2233'),
   315→   315→    });
   316→   316→    earthGroup.add(new THREE.Mesh(earthGeom, earthMat));
   317→   317→
   318→   318→    // Atmosphere glow (back-face sphere)
   319→   319→    const atmosGeom = new THREE.SphereGeometry(EARTH_RADIUS * 1.018, 64, 64);
   320→   320→    const atmosMat = new THREE.ShaderMaterial({
   321→   321→      vertexShader: ATMO_VS,
   322→   322→      fragmentShader: ATMO_FS,
   323→   323→      transparent: true,
   324→   324→      side: THREE.BackSide,
   325→   325→      depthWrite: false,
   326→   326→    });
   327→   327→    scene.add(new THREE.Mesh(atmosGeom, atmosMat));
   328→   328→
   329→   329→    // Markers group (child of earthGroup so markers rotate with the globe)
   330→   330→    const markersGroup = new THREE.Group();
   331→   331→    earthGroup.add(markersGroup);
   332→   332→    markersGroupRef.current = markersGroup;
   333→   333→
   334→   334→    // ── Animation loop ────────────────────────────────────────────────
   335→   335→    const clock = clockRef.current;
   336→   336→
   337→   337→    const animate = () => {
   338→   338→      animRef.current = requestAnimationFrame(animate);
   339→   339→      const elapsed = clock.getElapsedTime();
   340→   340→      const group = earthGroupRef.current;
   341→   341→      const cam = cameraRef.current;
   342→   342→      const ren = rendererRef.current;
   343→   343→      const mg = markersGroupRef.current;
   344→   344→      if (!group || !cam || !ren || !mg) return;
   345→   345→
   346→   346→      // Auto-rotate
   347→   347→      if (autoRotRef.current && !isDragRef.current && !zoomingToRef.current) {
   348→   348→        group.rotation.y += AUTO_SPEED;
   349→   349→      }
   350→   350→
   351→   351→      // Smooth zoom
   352→   352→      cam.position.z += (targetZoomRef.current - cam.position.z) * 0.06;
   353→   353→
   354→   354→      // Zoom-to-country animation
   355→   355→      if (zoomingToRef.current && targetRotYRef.current !== null) {
   356→   356→        let diff = targetRotYRef.current - group.rotation.y;
   357→   357→        while (diff > Math.PI) diff -= Math.PI * 2;
   358→   358→        while (diff < -Math.PI) diff += Math.PI * 2;
   359→   359→        group.rotation.y += diff * 0.06;
   360→   360→      }
   361→   361→
   362→   362→      // Nepal pulse ring
   363→   363→      const ring = nepalRingRef.current;
   364→   364→      if (ring) {
   365→   365→        const s = 1 + 0.5 * Math.sin(elapsed * 2.5);
   366→   366→        ring.scale.set(s, s, s);
   367→   367→        (ring.material as THREE.MeshBasicMaterial).opacity = 0.55 * (1 - (s - 1) / 0.5);
   368→   368→      }
